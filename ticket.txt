Overall, the latest commit looks good and now passes pytests locally.
I had vem/gemma-4 do quick evaluation of utils.training, and the only recommendation we should consider is this one:
"To improve the performance of compute_class_weights_tensor, you can replace the Python list accumulation with a running tensor count using torch.bincount. This avoids moving all labels to the CPU as Python integers and prevents the memory overhead of a potentially massive Python list.
Here is a mock-up of how that logic would be implemented:"
import torch
import numpy as np

def compute_class_weights_tensor_optimized(
    loader: torch.utils.data.DataLoader,
    num_classes: int,
    device: str = "cpu",
) -> torch.Tensor:
    """
    Optimized version of compute_class_weights_tensor using torch.bincount.
    """
    # Initialize a tensor to hold counts on the target device
    counts = torch.zeros(num_classes, dtype=torch.long, device=device)
    total_samples = 0

    # Iterate through the loader in batches
    for batch in loader:
        # Handle different batch formats (expecting (inputs, targets))
        _, labels = batch
        
        # Ensure labels are a 1D tensor on the correct device
        if not isinstance(labels, torch.Tensor):
            labels = torch.tensor(labels, dtype=torch.long, device=device)
        else:
            labels = labels.to(device).long().view(-1)

        # Update counts using bincount
        # minlength ensures the output tensor is at least num_classes long
        batch_counts = torch.bincount(labels, minlength=num_classes)
        
        # bincount might return a tensor longer than num_classes if labels exceed num_classes-1
        # so we slice it to exactly num_classes
        counts += batch_counts[:num_classes]
        total_samples += labels.numel()

    # Avoid division by zero for missing classes
    counts = torch.clamp(counts, min=1)

    # Compute inverse frequency weights
    # formula: total_samples / (num_classes * class_count)
    weights = total_samples / (num_classes * counts)

    return weights.float().cpu()
